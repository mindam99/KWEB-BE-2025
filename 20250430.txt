CREATE TABLE users (
  ID VARCHAR(30) NOT NULL,
  PW VARCHAR(100) NOT NULL,
  nickname VARCHAR(50) NOT NULL,
  profileLink VARCHAR(255),
  profileMessage VARCHAR(255),
  isExit TINYINT(1) DEFAULT 0,
  adm_Yr INT NOT NULL,
  PRIMARY KEY (ID)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

CREATE TABLE channels (
  id VARCHAR(30) NOT NULL,
  chName VARCHAR(50) NOT NULL,
  maker VARCHAR(30) NOT NULL,
  chLink VARCHAR(255),
  maxCapa INT,
  chExit TINYINT(1) DEFAULT 0,
  madeDate DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  FOREIGN KEY (maker) REFERENCES users(ID)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

CREATE TABLE chats (
  id VARCHAR(30) NOT NULL,
  messages TEXT NOT NULL,
  author VARCHAR(30) NOT NULL,
  chatCh VARCHAR(30) NOT NULL,
  chatStartDate DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  FOREIGN KEY (author) REFERENCES users(ID),
  FOREIGN KEY (chatCh) REFERENCES channels(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

CREATE TABLE follows (
  follower VARCHAR(30) NOT NULL,
  followee VARCHAR(30) NOT NULL,
  followDate DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (follower, followee),
  FOREIGN KEY (follower) REFERENCES users(ID),
  FOREIGN KEY (followee) REFERENCES users(ID)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

CREATE TABLE blocks (
  block VARCHAR(30) NOT NULL,
  blocked VARCHAR(30) NOT NULL,
  blockDate DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (block, blocked),
  FOREIGN KEY (block) REFERENCES users(ID),
  FOREIGN KEY (blocked) REFERENCES users(ID)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

Exercise 5.2
1.	SELECT users.id, users.name, tickets.seat_number
	FROM tickets
	JOIN users ON tickets.user = users.id
	WHERE tickets.train = 11
	ORDER BY tickets.seat_number;

2.	SELECT users.id, users.name, COUNT(DISTINCT tickets.train) AS trains_count, SUM(trains.distance) AS total_distance
	FROM tickets
	JOIN users ON tickets.user = users.id
	JOIN trains ON tickets.train = trains.id
	GROUP BY users.id, users.name
	ORDER BY total_distance DESC
	LIMIT 0, 6;

3.	SELECT trains.id, trains.type, src.name AS src_stn, dst.name AS dst_stn, TIMEDIFF(trains.arrival, trains.departure) AS travel_time
	FROM trains
	JOIN stations AS src ON trains.source = src.id
	JOIN stations AS dst ON trains.destination = dst.id
	ORDER BY travel_time DESC
	LIMIT 0, 6;

4.	SELECT trains.type, src.name AS src_stn, dst.name AS dst_stn, trains.departure, trains.arrival, ROUND((trains.distance / 10.0) * (types.fare_rate / 100.0), -2) AS fare
	FROM trains
	JOIN stations AS src ON trains.source = src.id
	JOIN stations AS dst ON trains.destination = dst.id
	JOIN types ON trains.type = types.id
	ORDER BY trains.departure

5.	SELECT trains.id, trains.type, src.name AS src_stn, dst.name AS dst_stn, COUNT(tickets.id) AS occupied, types.max_seats AS maximum
	FROM tickets
	JOIN trains ON tickets.train = trains.id
	JOIN types ON trains.type = types.id
	JOIN stations AS src ON trains.source = src.id
	JOIN stations AS dst ON trains.destination = dst.id
	GROUP BY trains.id, trains.type, src.name, dst.name, types.max_seats
	ORDER BY trains.id

6.	SELECT trains.id, trains.type, src.name AS src_stn, dst.name AS dst_stn, COUNT(tickets.id) AS occupied, types.max_seats AS maximum
	FROM trains
	LEFT JOIN tickets ON trains.id = tickets.train
	JOIN types ON trains.type = types.id
	JOIN stations AS src ON trains.source = src.id
	JOIN stations AS dst ON trains.destination = dst.id
	GROUP BY trains.id, trains.type, src.name, dst.name, types.max_seats
	ORDER BY trains.id
